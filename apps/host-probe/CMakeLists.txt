# Based on the CEF cefsimple macOS sample. See LICENSE-CEF.txt.
cmake_minimum_required(VERSION 3.21)
project(hitchhiker_host_probe LANGUAGES C CXX)
if(NOT APPLE)
  message(FATAL_ERROR "The host probe is macOS-only")
endif()
set(USE_SANDBOX ON CACHE BOOL "Enable CEF sandbox" FORCE)
set(PROJECT_ARCH arm64 CACHE STRING "Target architecture" FORCE)
if(NOT DEFINED CEF_ROOT OR NOT DEFINED NATIVE_SURFACE_LIBRARY)
  message(FATAL_ERROR "Use the build:native script to configure the pinned dependencies")
endif()
set(CMAKE_MODULE_PATH ${CMAKE_MODULE_PATH} "${CEF_ROOT}/cmake")
find_package(CEF REQUIRED)
add_subdirectory(${CEF_LIBCEF_DLL_WRAPPER_PATH} libcef_dll_wrapper)
include_directories(BEFORE "${CMAKE_CURRENT_SOURCE_DIR}")
list(APPEND CEF_COMPILER_FLAGS -fobjc-arc)
set(CEF_TARGET "hitchhiker-probe")
set(CEF_HELPER_TARGET "hitchhiker-probe_Helper")
set(CEF_HELPER_OUTPUT_NAME "hitchhiker-probe Helper")
SET_CEF_TARGET_OUT_DIR()
set(CEF_APP "${CEF_TARGET_OUT_DIR}/${CEF_TARGET}.app")
set(EXECUTABLE_NAME "${CEF_TARGET}")
set(PRODUCT_NAME "${CEF_TARGET}")
set(PROBE_RESOURCES mac/English.lproj/MainMenu.xib mac/English.lproj/InfoPlist.strings mac/cefsimple.icns)
add_executable(${CEF_TARGET} MACOSX_BUNDLE
  src/simple_app.cc src/simple_handler.cc src/simple_handler_mac.mm
  src/cefsimple_mac.mm src/native_sidebar.mm src/before_unload_dialog.mm src/page_manager.cc src/host_smoke_test.cc src/engine_bridge.cc ${PROBE_RESOURCES})
SET_EXECUTABLE_TARGET_PROPERTIES(${CEF_TARGET})
add_dependencies(${CEF_TARGET} libcef_dll_wrapper)
target_link_libraries(${CEF_TARGET} libcef_dll_wrapper ${CEF_STANDARD_LIBS} "${NATIVE_SURFACE_LIBRARY}")
target_compile_definitions(${CEF_TARGET} PRIVATE HITCHHIKER_PROFILE_PATH="${CMAKE_BINARY_DIR}/profile")
add_executable(browser_generation_test src/browser_generation_test.cc)
target_include_directories(browser_generation_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
set_target_properties(${CEF_TARGET} PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/mac/Info.plist.in")
COPY_MAC_FRAMEWORK("${CEF_TARGET}" "${CEF_BINARY_DIR}" "${CEF_APP}")
foreach(_suffix_list ${CEF_HELPER_APP_SUFFIXES})
  string(REPLACE ":" ";" _suffix_list ${_suffix_list})
  list(GET _suffix_list 0 _name_suffix)
  list(GET _suffix_list 1 _target_suffix)
  list(GET _suffix_list 2 _plist_suffix)
  set(_helper_target "${CEF_HELPER_TARGET}${_target_suffix}")
  set(_helper_output_name "${CEF_HELPER_OUTPUT_NAME}${_name_suffix}")
  set(_helper_info_plist "${CMAKE_CURRENT_BINARY_DIR}/helper-Info${_target_suffix}.plist")
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/mac/helper-Info.plist.in" _plist_contents)
  string(REPLACE "\${EXECUTABLE_NAME}" "${_helper_output_name}" _plist_contents "${_plist_contents}")
  string(REPLACE "\${PRODUCT_NAME}" "${_helper_output_name}" _plist_contents "${_plist_contents}")
  string(REPLACE "\${BUNDLE_ID_SUFFIX}" "${_plist_suffix}" _plist_contents "${_plist_contents}")
  file(WRITE "${_helper_info_plist}" "${_plist_contents}")
  add_executable(${_helper_target} MACOSX_BUNDLE src/process_helper_mac.cc)
  SET_EXECUTABLE_TARGET_PROPERTIES(${_helper_target})
  add_dependencies(${_helper_target} libcef_dll_wrapper)
  target_link_libraries(${_helper_target} libcef_dll_wrapper ${CEF_STANDARD_LIBS})
  set_target_properties(${_helper_target} PROPERTIES
    MACOSX_BUNDLE_INFO_PLIST "${_helper_info_plist}" OUTPUT_NAME "${_helper_output_name}")
  add_dependencies(${CEF_TARGET} ${_helper_target})
  add_custom_command(TARGET ${CEF_TARGET} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory
    "${CEF_TARGET_OUT_DIR}/${_helper_output_name}.app"
    "${CEF_APP}/Contents/Frameworks/${_helper_output_name}.app" VERBATIM)
endforeach()
COPY_MAC_RESOURCES("${PROBE_RESOURCES}" "mac/" "${CEF_TARGET}" "${CMAKE_CURRENT_SOURCE_DIR}" "${CEF_APP}")
PRINT_CEF_CONFIG()
